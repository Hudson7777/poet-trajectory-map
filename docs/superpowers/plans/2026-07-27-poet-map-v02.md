`projection.ts` 的 `buildTrajectoryPath` smooth 分支替换为（非 smooth 与空数组行为不变）：

```ts
export function buildTrajectoryPath(points: [number, number][], smooth = true): string {
  if (points.length === 0) return ''
  const fmt = (n: number) => Math.round(n * 100) / 100
  let d = `M${fmt(points[0][0])},${fmt(points[0][1])}`
  if (!smooth) {
    return d + points.slice(1).map(([x, y]) => `L${fmt(x)},${fmt(y)}`).join('')
  }
  for (let i = 0; i < points.length - 1; i++) {
    const p0 = points[i - 1] ?? points[i]
    const p1 = points[i]
    const p2 = points[i + 1]
    const p3 = points[i + 2] ?? p2
    const c1x = p1[0] + (p2[0] - p0[0]) / 6
    const c1y = p1[1] + (p2[1] - p0[1]) / 6
    const c2x = p2[0] - (p3[0] - p1[0]) / 6
    const c2y = p2[1] - (p3[1] - p1[1]) / 6
    d += `C${fmt(c1x)},${fmt(c1y)} ${fmt(c2x)},${fmt(c2y)} ${fmt(p2[0])},${fmt(p2[1])}`
  }
  return d
}
```

注意 fmt 会改变数字格式（整数不变，小数保留两位），`visibleStops`/`createProjection` 不动。

- [x] **Step 3: 实现**

`InkMap.tsx`：

```ts
export function zoomFilter(event: Event): boolean {
  if (event.type === 'wheel') {
    const we = event as WheelEvent
    return we.ctrlKey || we.metaKey
  }
  if (event.type.startsWith('touch')) return true
  return !(event as MouseEvent).button
}
```

zoom 行为加 `.filter(zoomFilter)`，并加 `.translateExtent([[0, 0], [vbW, vbH]])`（vbW/vbH 从 viewBox 解析）限制拖拽边界。
